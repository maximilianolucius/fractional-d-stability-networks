\section{Framework}
Let
\begin{equation}
{}^C D^\alpha x = F_G(x;\theta), \qquad 0<\alpha\le 1,
\end{equation}
where $G$ is a directed interaction graph. For an equilibrium $x^*$, write
\begin{equation}
J_G = D F_G(x^*).
\end{equation}
For commensurate order $\alpha$, Matignon's criterion requires
\begin{equation}
|\arg \lambda_i(J_G)|>\frac{\alpha\pi}{2}\qquad \text{for all }i.
\end{equation}

\begin{definition}[Purely fractional stabilization region]
For a graph $G$, define
\begin{equation}
\mathcal S_\alpha(G)=\left\{\theta:
|\arg\lambda_i(J_G)|>\frac{\alpha\pi}{2}\ \forall i,
\quad \max_i\Re\lambda_i(J_G)>0
\right\}.
\end{equation}
\end{definition}

The second condition excludes equilibria already Hurwitz stable in the first-order system.
